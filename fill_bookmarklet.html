Заполнение админки из Excel

Перетащи в закладки:

📋 Fill from Excel

Как использовать

  1. Запусти: python serve_data.py
  2. Открой карточку фильма в админке
  3. Нажми на закладку "📋 Fill from Excel"

Альтернатива — скопируй и вставь это в консоль (F12 → Console):

fetch('http://localhost:8766').then(r=>r.json()).then(d=>{
d.midrolls.forEach((v,i)=>{
if(i>0){let btn=document.querySelector('a[data-action="add-formset"]');if(btn)btn.click()}
let inp=document.getElementById(`id_middroll-${i}-time`);
if(inp){inp.value=v;inp.dispatchEvent(new Event('input',{bubbles:true}))}
});
let sel=document.querySelector('select[name$="-marker_type"]');
if(sel){sel.value='2';sel.dispatchEvent(new Event('change',{bubbles:true}));
let r=sel.closest('tr');
r.querySelector('input[name$="-start"]').value=d.start_scale;
r.querySelector('input[name$="-finish"]').value=d.finish_scale;}
let cr=document.getElementById('id_credits_begin_time');
if(cr){cr.value=d.postroll;cr.dispatchEvent(new Event('input',{bubbles:true}))}
let dur=document.getElementById('id_duration');
if(dur){dur.value=d.duration;dur.dispatchEvent(new Event('input',{bubbles:true}))}
alert('Done!');}).catch(e=>alert('Error: '+e.message));
